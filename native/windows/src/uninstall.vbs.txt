Option Explicit
Dim shl, fso, root, expected, marker, svc, ps, p, exe, desktop, programs, n
Set shl = CreateObject("WScript.Shell")
Set fso = CreateObject("Scripting.FileSystemObject")
root = fso.GetParentFolderName(WScript.ScriptFullName)
expected = shl.ExpandEnvironmentStrings("%LOCALAPPDATA%") & "\ChenJiangMagazine"
If LCase(fso.GetAbsolutePathName(root)) <> LCase(fso.GetAbsolutePathName(expected)) Then
  MsgBox "卸载程序位置不正确，请从开始菜单或系统“程序和功能”中卸载。", vbExclamation, "辰江网络杂志"
  WScript.Quit 1
End If
If Not fso.FileExists(root & "\.chenjiang-install") Then
  MsgBox "未找到安装标记，为保护文件，卸载已停止。", vbExclamation, "辰江网络杂志"
  WScript.Quit 1
End If
Set marker = fso.OpenTextFile(root & "\.chenjiang-install", 1)
If marker.ReadLine <> "ChenJiangMagazine/4/8d358ec6-1378-47fa-b2c6-527fae872420" Then
  marker.Close
  WScript.Quit 1
End If
marker.Close
If MsgBox("确定卸载辰江网络杂志吗？" & vbCrLf & vbCrLf & "将关闭杂志并删除程序。阅读记录和字号设置保留，重新安装后仍可使用。", vbYesNo + vbQuestion, "卸载辰江网络杂志") <> vbYes Then WScript.Quit 0
On Error Resume Next
Set svc = GetObject("winmgmts:\\.\root\cimv2")
Set ps = svc.ExecQuery("SELECT ExecutablePath FROM Win32_Process WHERE Name='ChenJiangMagazine.exe'")
For Each p In ps
  exe = ""
  If Not IsNull(p.ExecutablePath) Then exe = LCase(CStr(p.ExecutablePath))
  If Left(exe, Len(root) + 1) = LCase(root & "\") Then p.Terminate
Next
WScript.Sleep 1600
desktop = shl.SpecialFolders("Desktop")
If fso.FileExists(desktop & "\辰江网络杂志.lnk") Then fso.DeleteFile desktop & "\辰江网络杂志.lnk", True
programs = shl.SpecialFolders("Programs") & "\辰江网络杂志"
If fso.FileExists(programs & "\辰江网络杂志.lnk") Then fso.DeleteFile programs & "\辰江网络杂志.lnk", True
If fso.FileExists(programs & "\卸载辰江网络杂志.lnk") Then fso.DeleteFile programs & "\卸载辰江网络杂志.lnk", True
If fso.FolderExists(programs) Then
  If fso.GetFolder(programs).Files.Count = 0 And fso.GetFolder(programs).SubFolders.Count = 0 Then fso.DeleteFolder programs
End If
shl.RegDelete "HKCU\Software\Microsoft\Windows\CurrentVersion\Uninstall\ChenJiangMagazine\"
Err.Clear
fso.DeleteFolder root, True
n = Err.Number
On Error GoTo 0
If n = 0 Then
  MsgBox "已卸载辰江网络杂志。", vbInformation, "卸载完成"
Else
  MsgBox "程序仍有文件被占用。请关闭杂志后再运行一次卸载。", vbExclamation, "卸载提示"
End If
